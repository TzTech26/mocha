User-agent: *
Allow: /

# Proxied pages, the viewer that frames them, and game assets fetched from
# somebody else's CDN are all third party content served under this domain.
# Indexed, they read as scraped duplicates of other people's pages, which is
# what "low value content" means to a reviewer looking at this site.
Disallow: /~/
Disallow: /route/
Disallow: /cdn

# Artwork is the exception. Every game page carries the game's own picture and
# names it in its structured data, and a games result is worth a great deal
# less without one, so the images under /cdn are let through while the pages
# and the code beside them stay refused.
Allow: /cdn/*.png$
Allow: /cdn/*.jpg$
Allow: /cdn/*.jpeg$
Allow: /cdn/*.gif$
Allow: /cdn/*.webp$
Allow: /cdn/*.svg$
